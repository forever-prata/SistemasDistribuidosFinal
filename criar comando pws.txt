function Get-LastSale {
    $response = Invoke-WebRequest -Uri "http://localhost:8545" -Method Post -Headers @{"Content-Type"="application/json"} -Body '{"jsonrpc":"2.0","method":"eth_getBlockByNumber","params":["latest",true],"id":1}'
    $tx = ($response.Content | ConvertFrom-Json).result.transactions[-1]
    
    $hexData = $tx.input.Substring(2)
    $bytes = for ($i = 0; $i -lt $hexData.Length; $i += 2) {
        [Convert]::ToByte($hexData.Substring($i, 2), 16)
    }
    
    $tx | Format-List hash, from, blockNumber
    [System.Text.Encoding]::UTF8.GetString($bytes) | ConvertFrom-Json | Format-List
}
